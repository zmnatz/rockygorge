import home from "@content/home.yml";
import links from "@content/links.yml";
import forms from "@content/forms.yml";
import StoreItems from "@content/store.yml";
import events from "@content/events.yml";
import { Fragment } from "react";
import { Grid, Typography } from "@mui/material";
import { ProductCard } from "@/components/ProductCard";
import { CalendarCard } from "@/components/CalendarCard";
import { markdownToReact } from "@/utils/markdown";

export const visibleStoreItems = StoreItems.filter(item => !item.hide);
export const visibleEvents = events.filter(event => !event.hide);
export const visibleLinks = [...links, ...forms]
  .filter(item => item.header !== true && item.hide !== true);

export const sectionItems = {
  store: visibleStoreItems.map(item => ({
    key: item.slug,
    title: item?.description,
    href: `/${item.slug}`,
    summary: item.summary
  })),
  events: visibleEvents.map(event => ({
    key: event.slug,
    title: event.title,
    href: `/events/${event.slug}`,
    summary: event.summary
  })),
  links: visibleLinks.map(item => ({
    key: item.href,
    title: item.title,
    href: item.href,
    summary: item.summary
  }))
};

<Grid container spacing={2} sx={{ py: 2 }}>
  <Grid size={{xs: 12}} sx={{ textAlign: 'center' }}>
    {markdownToReact(home.hero.markdown)}
  </Grid>

  <Grid container spacing={2}>
    <Grid size={{ xs: 12, md: 6 }}>
      <Grid container spacing={2}>
        {home.sections.map((section) => (
          <Fragment key={section.source}>
            {section.title && (
              <Grid size={{ xs: 12 }}>
                <Typography variant="h5">{section.title}</Typography>
              </Grid>
            )}
            {sectionItems[section.source].map((item) => (
              <ProductCard key={item.key} title={item.title} href={item.href}>{item.summary}</ProductCard>
            ))}
          </Fragment>
        ))}
      </Grid>
    </Grid>
    <Grid size={{ xs: 12, md: 6 }}>
      <CalendarCard calendars={home.calendars} />
    </Grid>
  </Grid>
</Grid>
